// ========================================== // ฟังก์ชันบีบอัดภาพ (ปรับให้เบาขึ้น โหลดไวขึ้น) // ========================================== window.previewAndCompressImage = function(event) { const file = event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.readAsDataURL(file); reader.onload = function(e) { const img = new Image(); img.src = e.target.result; img.onload = function() { const canvas = document.createElement('canvas'); // ปรับขนาดความกว้างสูงสุดให้เล็กลงอีกนิด (200px ก็ชัดพอสำหรับรูปโปรไฟล์แล้วครับ) const MAX_WIDTH = 200; const MAX_HEIGHT = 200; let width = img.width; let height = img.height; if (width > height) { if (width > MAX_WIDTH) { height *= MAX_WIDTH / width; width = MAX_WIDTH; } } else { if (height > MAX_HEIGHT) { width *= MAX_HEIGHT / height; height = MAX_HEIGHT; } } canvas.width = width; canvas.height = height; const ctx = canvas.getContext('2d'); ctx.drawImage(img, 0, 0, width, height); // บีบอัดคุณภาพรูปเหลือ 60% (ลดขนาดไฟล์ลงเยอะมาก แต่ตาคนมองแทบไม่ออก) tempPhotoDataURL = canvas.toDataURL('image/jpeg', 0.6); document.getElementById('edit-profile-pic').src = tempPhotoDataURL; } } } // ========================================== // ระบบบันทึกรูปและชื่อ (เซฟพร้อมกันแบบ Parallel) // ========================================== document.getElementById('btn-save-profile').addEventListener('click', async () => { if (!activeUser) return; const newDisplayName = document.getElementById('edit-display-name').value; const currentLang = document.getElementById('btn-en').classList.contains('lang-active') ? 'en' : 'th'; const btnSave = document.getElementById('btn-save-profile'); const originalBtnHTML = btnSave.innerHTML; try { btnSave.innerHTML = ` ${translations[currentLang].saving}`; btnSave.disabled = true; // 1. สร้าง "ตระกร้างาน" เพื่อให้มันทำงานพร้อมกัน ไม่ต้องรอต่อคิว const updateTasks = []; // ถ้ามีการเปลี่ยนชื่อ ให้เพิ่มงานอัปเดตชื่อลงไป if (activeUser.displayName !== newDisplayName) { updateTasks.push(updateProfile(activeUser, { displayName: newDisplayName })); } // ถ้ามีการอัปโหลดรูปใหม่ ให้เพิ่มงานเซฟรูปลง Firestore ไปด้วย if (tempPhotoDataURL) { const userDocRef = doc(db, "users", activeUser.uid); updateTasks.push(setDoc(userDocRef, { profileImage: tempPhotoDataURL }, { merge: true })); } // 2. สั่งยิงข้อมูลทุกอย่าง "พร้อมกัน" (ประหยัดเวลาไปได้ 50%) if (updateTasks.length > 0) { await Promise.all(updateTasks); } // 3. อัปเดตหน้าจอทันที document.getElementById('home-display-name').textContent = newDisplayName; if (tempPhotoDataURL) { document.getElementById('home-profile-pic').src = tempPhotoDataURL; } // ล้างค่ารูปชั่วคราวทิ้งเผื่อกดเข้ามาแก้ใหม่ tempPhotoDataURL = null; // เด้งกลับหน้าแรกทันทีโดยไม่ต้องมี Alert มาขัดจังหวะ navigate('page-home'); } catch (error) { alert("เกิดข้อผิดพลาด: " + error.message); } finally { // คืนค่าปุ่มกลับสภาพเดิม btnSave.innerHTML = originalBtnHTML; btnSave.disabled = false; } });
