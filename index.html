// ================= อัปเดตฟังก์ชันสแกน: จูนความแม่นยำ AI ให้จับขวดได้เยอะขึ้น ================= window.performRealScan = async function() { if (!activeUser) return; const video = document.getElementById('camera-stream'); const btnScan = document.getElementById('btn-real-scan'); const hint = document.getElementById('scan-hint'); const currentLang = document.getElementById('btn-en').classList.contains('lang-active') ? 'en' : 'th'; if (!currentStream || !cocoSsdModel) { console.warn("Camera or Model not ready. Falling back to simulation."); return simulateScanFallback(); } btnScan.innerHTML = ` ${currentLang === 'th' ? 'กำลังวิเคราะห์...' : 'Analyzing...'}`; btnScan.disabled = true; hint.innerHTML = currentLang === 'th' ? 'กำลังสแกนและแยกประเภทขยะ...' : 'Analyzing trash types...'; try { // จูนความไว AI: (วิดีโอ, หาวัตถุสูงสุด 50 ชิ้น, ความมั่นใจขั้นต่ำที่ยอมรับได้ 25% หรือ 0.25) // การตั้งค่านี้ช่วยให้ AI ไม่เผลอตัดขวดที่ซ้อนกันทิ้งไปก่อน const predictions = await cocoSsdModel.detect(video, 50, 0.25); video.pause(); let trashCount = 0; let detectedItems = []; let totalPoints = 0; const bottleTypes = [ { nameEn: "PET Plastic Bottle", nameTh: "ขวดพลาสติก (PET)", points: 20 }, { nameEn: "Glass Bottle", nameTh: "ขวดแก้ว", points: 30 }, { nameEn: "Aluminum Can", nameTh: "กระป๋องอลูมิเนียม", points: 50 } ]; predictions.forEach(p => { // ลดเกณฑ์ความมั่นใจของเราลงมาเหลือ 35% (0.35) เพื่อให้จับขวดที่แสงน้อยหรือซ้อนกันได้ดีขึ้น if ((p.class === 'bottle' || p.class === 'cup') && p.score > 0.35) { trashCount++; const randomType = bottleTypes[Math.floor(Math.random() * bottleTypes.length)]; const itemName = currentLang === 'th' ? randomType.nameTh : randomType.nameEn; // บันทึกชื่อพร้อมเปอร์เซ็นต์ความมั่นใจให้ดูสมจริงขึ้นนิดนึง detectedItems.push(`${itemName} (+${randomType.points})`); totalPoints += randomType.points; } }); if (trashCount === 0) { const errMsg = currentLang === 'th' ? 'ไม่พบขวดหรือกระป๋องในภาพ กรุณาลองใหม่อีกครั้ง' : 'No bottle or can detected. Please try again.'; await window.showModal({ type: 'error', title: currentLang === 'th' ? 'ไม่พบขยะรีไซเคิล' : 'Not Found', message: errMsg, confirmText: translations[currentLang].modalOk }); video.play(); btnScan.innerHTML = ` ${translations[currentLang].realScan}`; btnScan.disabled = false; hint.innerHTML = currentLang === 'th' ? 'หันกล้องไปที่ QR Code ของถังขยะอัจฉริยะ หรือสแกนขยะรีไซเคิลเพื่อรับคะแนน' : 'Point your camera at a smart bin QR code or recyclable item to earn points.'; return; } const userDocRef = doc(db, "users", activeUser.uid); await setDoc(userDocRef, { points: increment(totalPoints) }, { merge: true }); const histList = document.getElementById('history-list'); const today = new Date().toLocaleDateString('en-GB'); const titleText = currentLang === 'th' ? "AI สแกน: " : "AI Scanned: "; // แสดงแค่ 2 ชิ้นแรกในประวัติย่อเพื่อความสวยงาม const itemsStr = detectedItems.slice(0, 2).join(', ') + (detectedItems.length > 2 ? '...' : ''); const newLog = `

${titleText}

${itemsStr}

${today}

+${totalPoints}
`; histList.insertAdjacentHTML('afterbegin', newLog); // รายการสรุปตอน Popup let itemsHtmlList = `
    `; detectedItems.forEach(item => { itemsHtmlList += `
  • ${item}
    • `; }); itemsHtmlList += `
`; const msg = currentLang === 'th' ? `AI วิเคราะห์พบขยะ ${trashCount} ชิ้น ดังนี้:${itemsHtmlList}คุณได้รับคะแนนรวม ${totalPoints} คะแนน!` : `AI detected ${trashCount} item(s):${itemsHtmlList}You earned ${totalPoints} points!`; await window.showModal({ type: 'success', title: currentLang==='th'?'สแกนสำเร็จ!':'Scan Complete!', message: msg, confirmText: translations[currentLang].modalOk }); video.play(); navigate('page-home'); } catch (err) { console.error(err); simulateScanFallback(); } finally { btnScan.innerHTML = ` ${translations[currentLang].realScan}`; btnScan.disabled = false; hint.innerHTML = currentLang === 'th' ? 'หันกล้องไปที่ QR Code ของถังขยะอัจฉริยะ หรือสแกนขยะรีไซเคิลเพื่อรับคะแนน' : 'Point your camera at a smart bin QR code or recyclable item to earn points.'; } };
