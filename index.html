window.performRealScan = async function() { if (!activeUser) return; const video = document.getElementById('camera-stream'); const btnScan = document.getElementById('btn-real-scan'); const hint = document.getElementById('scan-hint'); const currentLang = document.getElementById('btn-en').classList.contains('lang-active') ? 'en' : 'th'; if (!currentStream || !cocoSsdModel) { console.warn("Camera or Model not ready. Falling back to simulation."); return simulateScanFallback(); } btnScan.innerHTML = ` ${currentLang === 'th' ? 'กำลังวิเคราะห์...' : 'Analyzing...'}`; btnScan.disabled = true; hint.innerHTML = currentLang === 'th' ? 'กำลังส่งภาพให้ AI ประมวลผล...' : 'Sending image to AI...'; try { const predictions = await cocoSsdModel.detect(video); video.pause(); let trashCount = 0; let detectedItems = []; let totalPoints = 0; predictions.forEach(p => { // เช็คเงื่อนไข: ต้องเป็น 'bottle' เท่านั้น และความมั่นใจ > 40% if (p.class === 'bottle' && p.score > 0.40) { trashCount++; detectedItems.push(p.class); totalPoints += Math.floor(p.score * 30) + 10; } }); // ถ้าไม่พบขวดเลย ให้แสดงแจ้งเตือน Error และไม่แจกแต้ม if (trashCount === 0) { const errMsg = currentLang === 'th' ? 'ไม่พบขวดในภาพ กรุณาลองใหม่อีกครั้ง' : 'No bottle detected. Please try again.'; await window.showModal({ type: 'error', title: currentLang === 'th' ? 'ไม่พบวัตถุเป้าหมาย' : 'Not Found', message: errMsg, confirmText: translations[currentLang].modalOk }); video.play(); btnScan.innerHTML = ` ${translations[currentLang].realScan}`; btnScan.disabled = false; hint.innerHTML = currentLang === 'th' ? 'หันกล้องไปที่ QR Code ของถังขยะอัจฉริยะ หรือสแกนขยะรีไซเคิลเพื่อรับคะแนน' : 'Point your camera at a smart bin QR code or recyclable item to earn points.'; return; // หยุดการทำงานตรงนี้ ไม่เซฟลง Database } // ถ้าเจอขวด ให้เซฟคะแนนลง Database ตามปกติ const userDocRef = doc(db, "users", activeUser.uid); await setDoc(userDocRef, { points: increment(totalPoints) }, { merge: true }); const histList = document.getElementById('history-list'); const today = new Date().toLocaleDateString('en-GB'); const titleText = currentLang === 'th' ? "AI วิเคราะห์ภาพ: " : "AI Vision: "; const itemsStr = detectedItems.slice(0, 3).join(', ') + (detectedItems.length > 3 ? '...' : ''); const newLog = `

${titleText}${itemsStr}

${today}

+${totalPoints} POINT
`; histList.insertAdjacentHTML('afterbegin', newLog); const msg = currentLang === 'th' ? `AI ตรวจพบขวด ${trashCount} ใบ

คุณได้รับเพิ่ม ${totalPoints} คะแนน!` : `AI detected ${trashCount} bottle(s)

You earned ${totalPoints} points!`; await window.showModal({ type: 'success', title: currentLang==='th'?'สแกนสำเร็จ!':'Scan Complete!', message: msg, confirmText: translations[currentLang].modalOk }); video.play(); navigate('page-home'); } catch (err) { console.error(err); simulateScanFallback(); } finally { btnScan.innerHTML = ` ${translations[currentLang].realScan}`; btnScan.disabled = false; hint.innerHTML = currentLang === 'th' ? 'หันกล้องไปที่ QR Code ของถังขยะอัจฉริยะ หรือสแกนขยะรีไซเคิลเพื่อรับคะแนน' : 'Point your camera at a smart bin QR code or recyclable item to earn points.'; } };
