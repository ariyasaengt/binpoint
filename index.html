// ========================================== // อัปเดต: ระบบบันทึกรูปฟรี 100% ผ่าน ImgBB API // ========================================== document.getElementById('btn-save-profile').addEventListener('click', async () => { if (!activeUser) return; const newDisplayName = document.getElementById('edit-display-name').value; const currentLang = document.getElementById('btn-en').classList.contains('lang-active') ? 'en' : 'th'; const btnSave = document.getElementById('btn-save-profile'); const originalBtnHTML = btnSave.innerHTML; try { // เปลี่ยนปุ่มเป็นสถานะกำลังโหลด btnSave.innerHTML = ` ${translations[currentLang].saving}`; btnSave.disabled = true; let finalPhotoURL = activeUser.photoURL; // ถ้ายูสเซอร์มีการเลือกรูปภาพใหม่ if (tempPhotoDataURL) { // 1. ตัดส่วนหัวข้อมูลภาพออก ให้เหลือแค่เนื้อหาไฟล์ const base64Data = tempPhotoDataURL.split(',')[1]; // 2. เตรียมข้อมูลส่งไปที่เซิร์ฟเวอร์ ImgBB const formData = new FormData(); formData.append('image', base64Data); // 🔑 นำ API Key ที่ได้จากเว็บ ImgBB มาใส่ในเครื่องหมายคำพูดด้านล่างนี้ const imgbbApiKey = 'ใส่_API_KEY_ของคุณตรงนี้'; // 3. ยิงข้อมูลไปฝากไว้ที่ ImgBB (ฟรี) const response = await fetch(`https://api.imgbb.com/1/upload?key=${imgbbApiKey}`, { method: 'POST', body: formData }); const data = await response.json(); if (data.success) { // 4. ได้ลิงก์ URL รูปสั้นๆ กลับมาแล้ว! finalPhotoURL = data.data.url; } else { throw new Error("อัปโหลดรูปไม่สำเร็จ โปรดลองอีกครั้ง"); } } // 5. บันทึกชื่อและลิงก์รูปลงใน Firebase Auth ปกติ await updateProfile(activeUser, { displayName: newDisplayName, photoURL: finalPhotoURL }); alert(translations[currentLang].saveSuccess); // อัปเดตหน้าจอให้เป็นรูปใหม่ทันที document.getElementById('home-display-name').textContent = newDisplayName; if (finalPhotoURL) document.getElementById('home-profile-pic').src = finalPhotoURL; navigate('page-home'); } catch (error) { alert("เกิดข้อผิดพลาด: " + error.message); } finally { // คืนค่าปุ่มกลับเป็นเหมือนเดิม btnSave.innerHTML = originalBtnHTML; btnSave.disabled = false; } });
